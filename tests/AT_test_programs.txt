// ============================================================
// AtomC Type Analysis (AT) — Error Test Programs
// One program per semantic rule from "AtomC - analiza de tipuri"
// Use testat.c as the base valid program; each test violates
// exactly one rule. Run each block separately.
// ============================================================

// Base valid program (testat.c) — should produce NO errors:
// struct S{ int n; char text[16]; };
// struct S a; struct S v[10];
// void f(char text[],int i,char ch){ text[i]=ch; }
// int h(int x,int y){ if(x>0&&x<y){ f(v[x].text,y,'#'); return 1; } return 0; }


// ────────────────────────────────────────────────────────────
// AT TEST 1 — stm / IF
// Rule:  "the if condition must be a scalar value"
// Cause: using a struct as the if condition (structs are not scalar)
// ────────────────────────────────────────────────────────────
void doi() 
{
    int a;
}

void test1() {
    if (doi()) {}
}


// ────────────────────────────────────────────────────────────
// AT TEST 2 — stm / WHILE
// Rule:  "the while condition must be a scalar value"
// Cause: using a struct as the while condition
// ────────────────────────────────────────────────────────────
void doi() 
{
    int a;
}

void test2() {
    while (doi()) {}
}


// ────────────────────────────────────────────────────────────
// AT TEST 3 — stm / RETURN — void function returns a value
// Rule:  "a void function cannot return a value"
// Cause: returning an expression from a void function
// ────────────────────────────────────────────────────────────
void test3() {
    return 1;
}


// ────────────────────────────────────────────────────────────
// AT TEST 4 — stm / RETURN — non-void function returns nothing
// Rule:  "a non-void function must return a value"
// Cause: return with no expression in a non-void function
// ────────────────────────────────────────────────────────────
int test4() {
    return;
}


// ────────────────────────────────────────────────────────────
// AT TEST 5 — stm / RETURN — return value not scalar
// Rule:  "the return value must be a scalar value"
// Cause: returning a struct (structs are not scalar)
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S mk() {
    struct S a;
    return a;
}


// ────────────────────────────────────────────────────────────
// AT TEST 6 — stm / RETURN — incompatible return type
// Rule:  "cannot convert the return expression type to the function return type"
// Cause: trying to return a struct S where a char is expected
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

char test6() {
    return a;
}


// ────────────────────────────────────────────────────────────
// AT TEST 7 — exprAssign — destination not left-value
// Rule:  "the assign destination must be a left-value"
// Cause: assigning to a literal (1 is an rval)
// ────────────────────────────────────────────────────────────
void test7() {
    1 = 5;
}


// ────────────────────────────────────────────────────────────
// AT TEST 8 — exprAssign — destination is constant
// Rule:  "the assign destination cannot be constant"
// Cause: assigning to an array variable (arrays are ct=true in AT)
// ────────────────────────────────────────────────────────────
int v[10];

void test8() {
    v = v;
}


// ────────────────────────────────────────────────────────────
// AT TEST 9 — exprAssign — destination not scalar
// Rule:  "the assign destination must be scalar"
// Cause: assigning to a struct variable (structs are not scalar)
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;
struct S b;

void test9() {
    a = b;
}


// ────────────────────────────────────────────────────────────
// AT TEST 10 — exprAssign — source not convertible to destination
// Rule:  "the assign source cannot be converted to destination"
// Cause: assigning a struct S1 to a struct S2 variable (different struct types)
// ────────────────────────────────────────────────────────────
struct S1 { int x; };
struct S2 { int y; };
struct S1 a;
struct S2 b;

void test10() {
    a.x = b.y;
}


// ────────────────────────────────────────────────────────────
// AT TEST 11 — exprOr — invalid operand type for ||
// Rule:  "invalid operand type for ||"
// Cause: using a struct as an operand of ||
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

void test11() {
    int x;
    x = a || 1;
}


// ────────────────────────────────────────────────────────────
// AT TEST 12 — exprAnd — invalid operand type for &&
// Rule:  "invalid operand type for &&"
// Cause: using a struct as an operand of &&
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

void test12() {
    int x;
    x = 1 && a;
}


// ────────────────────────────────────────────────────────────
// AT TEST 13 — exprEq — invalid operand type for == or !=
// Rule:  "invalid operand type for == or !="
// Cause: comparing two structs with ==
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;
struct S b;

void test13() {
    int x;
    x = a == b;
}


// ────────────────────────────────────────────────────────────
// AT TEST 14 — exprRel — invalid operand type for <, <=, >, >=
// Rule:  "invalid operand type for <, <=, >, >="
// Cause: comparing two structs with <
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;
struct S b;

void test14() {
    int x;
    x = a < b;
}


// ────────────────────────────────────────────────────────────
// AT TEST 15 — exprAdd — invalid operand type for + or -
// Rule:  "invalid operand type for + or -"
// Cause: adding two arrays (no pointer arithmetic in AtomC)
// ────────────────────────────────────────────────────────────
int v[10];
int w[10];

void test15() {
    int x;
    x = v + w;
}


// ────────────────────────────────────────────────────────────
// AT TEST 16 — exprMul — invalid operand type for * or /
// Rule:  "invalid operand type for * or /"
// Cause: multiplying two arrays
// ────────────────────────────────────────────────────────────
int v[10];
int w[10];

void test16() {
    int x;
    x = v * w;
}


// ────────────────────────────────────────────────────────────
// AT TEST 17 — exprCast — cannot convert to a struct type
// Rule:  "cannot convert to a struct type"
// Cause: casting to a struct type
// ────────────────────────────────────────────────────────────
struct S { int n; };

void test17() {
    int x;
    struct S s;
    s = (struct S)x;
}


// ────────────────────────────────────────────────────────────
// AT TEST 18 — exprCast — cannot convert a struct
// Rule:  "cannot convert a struct"
// Cause: casting a struct expression to int
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

void test18() {
    int x;
    x = (int)a;
}


// ────────────────────────────────────────────────────────────
// AT TEST 19 — exprCast — array converted to scalar
// Rule:  "an array can be converted only to another array"
// Cause: casting an array to a scalar type (int)
// ────────────────────────────────────────────────────────────
int v[10];

void test19() {
    int x;
    x = (int)v;
}


// ────────────────────────────────────────────────────────────
// AT TEST 20 — exprCast — scalar converted to array
// Rule:  "a scalar can be converted only to another scalar"
// Cause: casting a scalar int to an array type
// ────────────────────────────────────────────────────────────
void test20() {
    int x;
    int y;
    y = (int[])x;
}


// ────────────────────────────────────────────────────────────
// AT TEST 21 — exprUnary — non-scalar operand for unary -/!
// Rule:  "unary - or ! must have a scalar operand"
// Cause: applying unary - to a struct
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

void test21() {
    int x;
    x = -a;
}


// ────────────────────────────────────────────────────────────
// AT TEST 22 — exprPostfix — indexing a non-array
// Rule:  "only an array can be indexed"
// Cause: indexing a plain int variable
// ────────────────────────────────────────────────────────────
void test22() {
    int x;
    int y;
    y = x[0];
}


// ────────────────────────────────────────────────────────────
// AT TEST 23 — exprPostfix — index not convertible to int
// Rule:  "the index is not convertible to int"
// Cause: using a struct as an array index
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;
int v[10];

void test23() {
    int x;
    x = v[a];
}


// ────────────────────────────────────────────────────────────
// AT TEST 24 — exprPostfix — dot on non-struct
// Rule:  "a field can only be selected from a struct"
// Cause: using dot operator on an int
// ────────────────────────────────────────────────────────────
void test24() {
    int x;
    int y;
    y = x.n;
}


// ────────────────────────────────────────────────────────────
// AT TEST 25 — exprPostfix — field does not exist in struct
// Rule:  "the structure X does not have a field Y"
// Cause: accessing a field name that doesn't exist
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

void test25() {
    int x;
    x = a.doi;
}


// ────────────────────────────────────────────────────────────
// AT TEST 26 — exprPrimary — undefined identifier
// Rule:  "undefined id: X"
// Cause: using a variable that was never declared
// ────────────────────────────────────────────────────────────
void test26() {
    int x;
    x = undeclared;
}


// ────────────────────────────────────────────────────────────
// AT TEST 27 — exprPrimary — calling a non-function
// Rule:  "only a function can be called"
// Cause: calling a variable as if it were a function
// ────────────────────────────────────────────────────────────
void test27() {
    int x;
    x = x();
}


// ────────────────────────────────────────────────────────────
// AT TEST 28 — exprPrimary — using a function without calling it
// Rule:  "a function can only be called"
// Cause: using a function name as a value without ()
// ────────────────────────────────────────────────────────────
int foo() { return 1; }

void test28() {
    int x;
    x = foo;
}


// ────────────────────────────────────────────────────────────
// AT TEST 29 — exprPrimary — too many arguments in function call
// Rule:  "too many arguments in function call"
// Cause: passing more arguments than parameters
// ────────────────────────────────────────────────────────────
int add(int a, int b) { return a; }

void test29() {
    int x;
    x = add(1, 2, 3);
}


// ────────────────────────────────────────────────────────────
// AT TEST 30 — exprPrimary — too few arguments in function call
// Rule:  "too few arguments in function call"
// Cause: passing fewer arguments than parameters
// ────────────────────────────────────────────────────────────
int add(int a, int b) { return a; }

void test30() {
    int x;
    x = add(1);
}


// ────────────────────────────────────────────────────────────
// AT TEST 31 — exprPrimary — argument type not convertible to parameter type
// Rule:  "in call, cannot convert the argument type to the parameter type"
// Cause: passing a struct where an int is expected
// ────────────────────────────────────────────────────────────
struct S { int n; };
struct S a;

int takes_int(int x) { return x; }

void test31() {
    int x;
    x = takes_int(a);
}


// ────────────────────────────────────────────────────────────
// VALID PROGRAM — testat.c (should produce NO errors)
// Tests all the rules passing correctly:
//   - if/while conditions are scalar (int comparisons)
//   - return type matches function return type
//   - array indexing uses an int index on an array
//   - dot operator used on struct type, existing field
//   - function called with correct arg count and types
// ────────────────────────────────────────────────────────────
struct S {
    int n;
    char text[16];
};

struct S a;
struct S v[10];

void f(char text[], int i, char ch) {
    text[i] = ch;
}

int h(int x, int y) {
    if (x > 0 && x < y) {
        f(v[x].text, y, '#');
        return 1;
    }
    return 0;
}
