// testgc.c
int fact(int n){
    if(n<3)return n;
    return n*fact(n-1);
}

void main(){
    put_i(4.9);       // => 4
    put_i(fact(3));   // => 6
    int r;
    r=1;
    int i;
    i=2;
    while(i<5){
        r=r*i;
        i=i+1;
    }
    put_i(r);         // => 24
}

// ═══════════════════════════════════════════════════════════
// CADRUL FUNCTIEI fact(int n)
// ═══════════════════════════════════════════════════════════
//
//   -2       -1         0
//    n    adr_return  old_FP
//
// fact are 0 variabile locale, deci ENTER 0.
// n este parametru, accesat la FP[-2].

// ═══════════════════════════════════════════════════════════
// CADRUL FUNCTIEI main()
// ═══════════════════════════════════════════════════════════
//
//   -1         0       1       2
// adr_return old_FP   r       i
//
// main are 2 variabile locale (r si i), deci ENTER 2.
// main nu are parametri (apelul initial are 0 argumente).

// ═══════════════════════════════════════════════════════════
// CODUL GENERAT (pseudocod MV)
// ═══════════════════════════════════════════════════════════

// ─── Entry point generat de main.c ───────────────────────
CALL main           <- push adr(HALT), sare la ENTER
HALT                <- executia se opreste dupa return

// ─── void main() ──────────────────────────────────────────
main: ENTER 2       <- push old_FP, FP=SP, SP+=2 (r la FP[1], i la FP[2])

// put_i(4.9):
//   4.9 este DOUBLE, put_i asteapta INT -> insertConvIfNeeded emite CONV.f.i
PUSH.f 4.9          <- valoarea argumentului 4.9
CONV.f.i            <- converteste 4.9 -> 4 (int)
CALL_EXT put_i      <- tipareste 4, pops int
DROP                <- put_i e void, rezultatul expresiei e aruncat

// put_i(fact(3)):
//   mai intai se calculeaza argumentul fact(3)
PUSH.i 3            <- argumentul pentru fact
CALL fact           <- push adr(urm. instructiune), sare la fact
// ... (fact ruleaza recursiv, returneaza 6 pe stiva)
CALL_EXT put_i      <- tipareste 6
DROP

// r=1:
FPADDR.i 1          <- push adresa FP[1] (r)
PUSH.i 1            <- valoarea 1
STORE.i             <- *(int*)FP[1] = 1; lasa 1 pe stiva
DROP                <- rezultatul atribuirii e aruncat

// i=2:
FPADDR.i 2          <- push adresa FP[2] (i)
PUSH.i 2            <- valoarea 2
STORE.i             <- *(int*)FP[2] = 2; lasa 2 pe stiva
DROP

// while(i<5):
WHILE_COND:
FPADDR.i 2          <- push adresa i
LOAD.i              <- dereferentiaza: push valoarea lui i
PUSH.i 5            <- push 5
LESS.i              <- i < 5 -> int 0 sau 1
JF AFTER_WHILE      <- daca fals, iesi din bucla

// r=r*i:
FPADDR.i 1          <- push adresa r (destinatia)
FPADDR.i 1          <- push adresa r (pentru LOAD)
LOAD.i              <- push valoarea lui r
FPADDR.i 2          <- push adresa i (pentru LOAD)
LOAD.i              <- push valoarea lui i
MUL.i               <- r * i
STORE.i             <- *(int*)FP[1] = r*i; lasa rezultatul pe stiva
DROP

// i=i+1:
FPADDR.i 2          <- push adresa i (destinatia)
FPADDR.i 2          <- push adresa i (pentru LOAD)
LOAD.i              <- push valoarea lui i
PUSH.i 1            <- push 1
ADD.i               <- i + 1
STORE.i             <- *(int*)FP[2] = i+1
DROP

JMP WHILE_COND      <- inapoi la conditia while

AFTER_WHILE: NOP

// put_i(r):
FPADDR.i 1          <- push adresa r
LOAD.i              <- push valoarea lui r (24)
CALL_EXT put_i      <- tipareste 24
DROP

RET_VOID 0          <- return (0 parametri), restaureaza FP/SP/IP

// ─── int fact(int n) ──────────────────────────────────────
fact: ENTER 0       <- push old_FP, FP=SP, SP+=0 (fara variabile locale)

// if(n<3):
FPADDR.i -2         <- push adresa n (parametru, la FP[-2])
LOAD.i              <- push valoarea lui n
PUSH.i 3            <- push 3
LESS.i              <- n < 3 -> int
JF ELSE_BRANCH      <- daca fals (n>=3), sari la return n*fact(n-1)

// return n:
FPADDR.i -2         <- push adresa n
LOAD.i              <- push valoarea lui n
RET 1               <- return n; restaureaza frame, 1 parametru

ELSE_BRANCH: NOP

// return n*fact(n-1):
//   mai intai se calculeaza n (stanga MUL)
FPADDR.i -2         <- push adresa n (stanga inmultirii)
LOAD.i              <- push valoarea lui n
//   apoi se calculeaza fact(n-1) (dreapta MUL)
FPADDR.i -2         <- push adresa n (pentru scadere)
LOAD.i              <- push valoarea lui n
PUSH.i 1            <- push 1
SUB.i               <- n - 1
CALL fact           <- push ret addr, recursie cu n-1
// fact returneaza result pe stiva
MUL.i               <- n * fact(n-1)
RET 1               <- return rezultat; 1 parametru

// ═══════════════════════════════════════════════════════════
// TRASEUL RECURSIV fact(3)
// ═══════════════════════════════════════════════════════════
//
// fact(3): n=3, 3<3 -> fals -> n*fact(n-1) = 3*fact(2)
//   fact(2): n=2, 2<3 -> adevarat -> return 2
// 3 * 2 = 6
//
// Stiva la apelul fact(3):
//   [... | n=3 | ret | oldFP ]   <- ENTER 0
//   n<3 -> 0 -> JF trece
//   LOAD n=3, LOAD n=3, PUSH 1, SUB -> 2 pe stiva
//   CALL fact(2):
//     [... | n=3 | ret1 | oldFP1 | n=2 | ret2 | oldFP2 ]
//     2<3 -> adevarat -> LOAD n=2, RET 1 -> returneaza 2
//   Stiva: [... | n=3 | ret1 | oldFP1 | 3 | 2]
//   MUL.i: 3*2 = 6
//   RET 1 -> returneaza 6

// ═══════════════════════════════════════════════════════════
// OUTPUT ASTEPTAT
// ═══════════════════════════════════════════════════════════
//
// => 4      (put_i(4.9) -> CONV.f.i -> 4)
// => 6      (put_i(fact(3)) -> 3*2 = 6)
// => 24     (put_i(r) -> 1*2*3*4 = 24)
